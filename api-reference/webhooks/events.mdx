Every webhook uses the `eventId`, `eventType`, `data`, and `timestamp` envelope described in the [Webhooks Overview](/api-reference/webhooks/overview).

## Checkout Events

Checkout events are emitted when a Checkout Wallet reaches a terminal state. Use `data.reference` to reconcile the event to your order and `eventId` to deduplicate deliveries.

### checkout.completed

Fired after the exact `amountPayable` is confirmed and merchant settlement is recorded. This is the only checkout event on which you should fulfill an order.

```json
{
  "eventId": "evt_mtwc9i1y_0409d781714d8fa2",
  "eventType": "checkout.completed",
  "data": {
    "checkoutId": "35f5f6fb-0c8d-4e96-ba4a-c04750724ff6",
    "transactionId": "3469cd26-a213-431e-96c9-72dc2dea51ae",
    "reference": "ORDER-10025",
    "status": "completed",
    "amount": "5000.00",
    "fee": "80.00",
    "amountPayable": "5080.00",
    "settlementAmount": "5000.00",
    "feeBearer": "customer",
    "currency": "NGN",
    "accountType": "temporary",
    "sender": {
      "accountName": "ADA LOVELACE",
      "accountNumber": "0123456789",
      "bankCode": "100004",
      "bankName": "OPAY"
    },
    "providerReference": "100004260911021740170939156478",
    "metadata": { "orderId": "10025" },
    "expiresAt": "2026-09-11T10:30:00.000Z",
    "completedAt": "2026-09-11T10:08:14.148Z",
    "failedAt": null,
    "reversedAt": null
  },
  "timestamp": "2026-09-11T10:08:14.186Z"
}
```

`sender.bankCode` is exposed when the provider supplies a sender bank code. `sender.bankName` can be `null` when the upstream provider does not supply the bank name.

## Payout Events

Master-wallet payout events are emitted when a payout reaches a terminal state. Use `data.reference` to reconcile with your request and `data.payoutId` when retrieving the payout through the API.

### payout.completed

Fired when the payout completes successfully. A same-business transfer may produce this event immediately during the create request.

```json
{
  "eventId": "evt_payout_completed_001",
  "eventType": "payout.completed",
  "data": {
    "payoutId": "a9af907d-965f-4ad8-9472-f437fc11df21",
    "transactionId": "5a0bb85c-268b-44fa-a57d-70b2944688bc",
    "reference": "PAYOUT-10025",
    "status": "completed",
    "route": "external_bank",
    "amount": "5000.00",
    "fee": "80.00",
    "totalDebited": "5080.00",
    "currency": "NGN",
    "recipient": {
      "bankCode": "058",
      "bankName": "Guaranty Trust Bank",
      "accountName": "ADA LOVELACE",
      "accountNumber": "0123456789"
    },
    "providerReference": "PYT_K8Q4D2M7N5R3T9A_1789098000000",
    "metadata": { "invoiceId": "INV-10025" },
    "completedAt": "2026-09-11T12:05:10.000Z",
    "failedAt": null,
    "reversedAt": null

  "timestamp": "2026-09-11T12:05:10.100Z"

    "status": "reversed",
    "route": "external_bank",
    "amount": "5000.00",
    "fee": "80.00",
    "totalDebited": "5080.00",
    "currency": "NGN",
    "recipient": {
      "bankCode": "058",
      "bankName": "Guaranty Trust Bank",
      "accountName": "ADA LOVELACE",
      "accountNumber": "0123456789"
    },
    "providerReference": "PYT_K8Q4D2M7N5R3T9A_1789098000000",
    "metadata": { "invoiceId": "INV-10025" },
    "completedAt": null,
    "failedAt": "2026-09-11T12:05:10.000Z",
    "reversedAt": "2026-09-11T12:05:12.000Z"

```

The `route` value is one of `same_business`, `fossapay_cross_business`, or `external_bank`. Same-business payouts have a zero fee.

## Example Handler

```javascript
app.post('/webhooks/fossapay', async (req, res) => {
  const { eventId, eventType, data } = req.body;

  res.status(200).send('OK');

  // Deduplicate eventId before running any business action.
  switch (eventType) {
    case 'checkout.completed':
      await handleCheckoutCompleted(data);
      break;

    case 'checkout.failed':
    case 'checkout.expired':
    case 'checkout.reversed':
      await handleCheckoutTerminalState(eventType, data);
      break;

    case 'payout.completed':
      await handlePayoutCompleted(data);
      break;

    case 'payout.failed':
    case 'payout.reversed':
      await handlePayoutTerminalState(eventType, data);
      break;

    case 'deposit.completed':
      await handleDepositCompleted(data);
      break;

    case 'withdrawal.completed':
      await handleWithdrawalCompleted(data);
      break;

    case 'transfer.completed':
      await handleTransferCompleted(data);
      break;

    case 'transfer.failed':
      await handleTransferFailed(data);
      break;

    case 'wallet.balance.updated':
      await handleWalletBalanceUpdated(data);
      break;

    default:
      console.log('Unhandled event:', eventType, eventId);
  }
});
```
